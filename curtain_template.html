/*__DATA__*/ /* ---------- fixed structure ---------- */ // STRAND_GAP against the widest bloom is what decides whether this reads as a // curtain of garlands or as a scatter of columns. // There is no single hanging line: strands start at the arch soffit above // them, so each carries its own `top` and depths are measured from them. const STRAND_GAP = 27, MARGIN = 30, PAD_W = 9; // A bloom is round, so one number is both its width and its height — word count // now reads as how open the flower is rather than how long a bead is. Growth is // fast then levels off, so a 60-word prompt is a full head and a 3-word one is // a tight bud without the longest prompt dwarfing the rest. const D_MIN = 11, D_MAX = 31, D_K = 17; const KIND_BUD = 0, KIND_SMALL = 1, KIND_FULL = 2, KIND_DOUBLE = 3; const KIND_NAMES = ['bud','half-open','full bloom','double bloom']; function bloomSize(wc){ return D_MIN + (D_MAX - D_MIN) * (1 - Math.exp(-wc / D_K)); } function bloomKind(wc){ return wc <= 5 ? KIND_BUD : wc <= 15 ? KIND_SMALL : wc <= 40 ? KIND_FULL : KIND_DOUBLE; } function beadGap(g){ return g < 0 ? 0 : 1.5 + 8 * (1 - Math.exp(-g / 1000)); } // widest bloom, for layout and hit testing const D_WIDEST = bloomSize(1e4); function rnd(seed){ let x = Math.sin(seed * 12.9898) * 43758.5453; return x - Math.floor(x); } // Paper tooth, laid over the finished frame. Only the width is needed here — the floor comes later, once the longest strand is known. */ const ARCH = (() => { const padX = 38, headroom = 54; const inL = -padX, inR = worldW + padX; const openW = inR - inL, a = openW/2; const spring = -headroom; // where the arch leaves the jamb // Classic two-centred pointed arch: each half is struck from a centre offset // past the middle, and the two meet in a point at the crown.
